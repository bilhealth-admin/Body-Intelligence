from pathlib import Path
import datetime as dt
import hashlib
import json
import os
import subprocess
import sys
import time

ROOT = Path('/workspace/scratch/ada7a97de764')
WORK = ROOT / 'validation'
OUT = ROOT / 'repository/docs/qa_parallel/bil04/final_validation'
FILES = sorted((WORK / 'test/parallel/bil04').glob('*_test.dart'))
started = time.monotonic()
reports = []
retained_reports = []
env = {**os.environ, 'TAR_OPTIONS': '--no-same-owner'}
for path in FILES:
    relative = path.relative_to(WORK).as_posix()
    name = path.stem
    # The prior source's reports are preserved in before_registry_contract_update.
    # Run every original file now; do not reuse any prior evidence for this gate.
    command = [str(ROOT / 'runtime/with-flutter'), 'test', '--frontend-server-starter-path=' + str(ROOT / 'runtime/compiler_heap_starter.dart'), '--no-pub', '--reporter', 'expanded', '--concurrency=1', relative]
    record = ['python3', 'tool/qa_parallel/bil04/record_validation.py', '--name', name, '--output', str(OUT / 'owned_files'), '--', *command]
    result = subprocess.run(record, cwd=WORK, env=env)
    if result.returncode != 0:
        print('Stopped at nonpassing file:', relative, flush=True)
        sys.exit(result.returncode)
    reports.append(json.loads((OUT / 'owned_files' / (name + '.json')).read_text()))

reference = reports[0]
digest = reference['source_digest_before']
assert all(r['result'] == 'PASS' and r['exit_code'] == 0 and r['source_stable']
           and r['source_digest_before'] == r['source_digest_after'] == digest
           and isinstance(r['passed_test_count'], int) for r in reports)
source_name = 'all_owned_files_final.sources.json'
(OUT / source_name).write_bytes((OUT / 'owned_files' / reference['source_manifest']).read_bytes())
summary = {
    'name': 'all_owned_files_final', 'result': 'PASS', 'exit_code': 0,
    'passed_test_count': sum(r['passed_test_count'] for r in reports),
    'test_file_count': len(FILES), 'single_process_run': False,
    'auxiliary_runtime_configuration': 'compiler_launch_configuration.json',
    'aggregation': 'All original owned *_test.dart files, one serial Flutter process per file. Counts are from disjoint files; no previous attempts or overlapping suites are added.',
    'command_argv': [r['command_argv'] for r in reports], 'cwd': str(WORK),
    'environment': reference['environment'],
    'orchestration_invocation_duration_seconds': round(time.monotonic() - started, 3),
    'summed_file_duration_seconds': round(sum(r['duration_seconds'] for r in reports), 3),
    'first_file_started_at_utc': min(r['started_at_utc'] for r in reports),
    'last_file_ended_at_utc': max(r['ended_at_utc'] for r in reports),
    'retained_exact_source_file_reports': retained_reports,
    'duration_note': 'Per-file durations are actual recorder times. The orchestrator duration describes this invocation only; any retained reports are listed explicitly. No single-process aggregate runtime is claimed.',
    'source_digest_algorithm': reference['source_digest_algorithm'],
    'source_scope': reference['source_scope'], 'source_digest_before': digest,
    'source_digest_after': digest, 'source_stable': True,
    'source_file_count': reference['source_file_count'], 'source_manifest': source_name,
    'file_reports': ['owned_files/' + r['name'] + '.json' for r in reports],
    'log': 'all_owned_files_final.txt', 'skips_or_exclusions_added': False,
}
summary['evidence_span_seconds'] = round((dt.datetime.fromisoformat(summary['last_file_ended_at_utc']) - dt.datetime.fromisoformat(summary['first_file_started_at_utc'])).total_seconds(), 6)
lines = [f"{r['name']}: PASS {r['passed_test_count']} tests, exit 0, source {digest}" for r in reports]
lines.append(f"Total: {summary['passed_test_count']} passing cases across {len(FILES)} independent serial file runs.")
(OUT / 'all_owned_files_final.txt').write_text('\n'.join(lines) + '\n')
summary['log_sha256'] = hashlib.sha256((OUT / summary['log']).read_bytes()).hexdigest()
(OUT / 'all_owned_files_final.json').write_text(json.dumps(summary, indent=2) + '\n')
print(json.dumps({k: summary[k] for k in ('result', 'passed_test_count', 'test_file_count', 'single_process_run', 'source_stable')}, indent=2), flush=True)
